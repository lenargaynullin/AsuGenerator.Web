@code {
    // UI-переменные селекторов экрана
    private string selectedCabinetType = "Шкаф управления вентиляцией (ШУВ)";
    private string enclosureType = "Навесной (в стену/на стену)";
    private string voltageType = "380В";
    private string heaterType = "Водяной";
    private decimal marginMultiplier = 1.35m;

    // Внутренние B2B-модели данных
    private VentAvtomatikaConfig? loadedConfig;
    private List<SelectedComponent> matchedComponents = new();
    private string svgPreview = string.Empty;

    // Инжектируем нашу фабрику стратегий
    [Inject] private CabinetStrategyFactory StrategyFactory { get; set; } = default!;

    private void ProcessUiCalculation()
    {
        // 1. Формируем b2b-контейнер параметров с экрана
        var input = new UiConfigInput
        {
            EnclosureType = enclosureType,
            VoltageType = voltageType,
            TechnologyType = heaterType,
            ProjectNumber = "Конфиг-" + DateTime.Now.ToString("ddMM")
        };

        // Поддерживаем совместимость со старым кодом штампа на экране
        loadedConfig = new VentAvtomatikaConfig
        {
            ClientName = "Прямой конфигуратор сайта",
            CompanyName = "B2B Заказчик",
            KpNumber = input.ProjectNumber,
            Heater1Type = $"{heaterType} нагреватель",
            HeaterEl1Voltage = voltageType
        };

        // 2. Получаем нужную стратегию через фабрику
        ICabinetStrategy currentStrategy = StrategyFactory.GetStrategy(selectedCabinetType);

        // 3. Делегируем расчет стратегии шкафа
        matchedComponents = currentStrategy.CalculateComponents(input);
    }

    private async Task HandleFileSelected(InputFileChangeEventArgs e)
    {
        try
        {
            using var stream = e.File.OpenReadStream(maxAllowedSize: 10 * 1024 * 1024);
            using var memoryStream = new MemoryStream();
            await stream.CopyToAsync(memoryStream);
            memoryStream.Position = 0;

            loadedConfig = Parser.ParseVentaFile(memoryStream);

            if (loadedConfig != null)
            {
                matchedComponents = Engine.RunB2bLogic(loadedConfig);
            }
        }
        catch (Exception ex)
        {
            System.Diagnostics.Debug.WriteLine($"Ошибка парсинга: {ex.Message}");
        }
    }

    private async Task DownloadFile()
    {
        if (loadedConfig == null || matchedComponents == null || !matchedComponents.Any()) return;

        try
        {
            var input = new UiConfigInput { EnclosureType = enclosureType, VoltageType = voltageType, TechnologyType = heaterType, ProjectNumber = loadedConfig.KpNumber };
            ICabinetStrategy currentStrategy = StrategyFactory.GetStrategy(selectedCabinetType);

            byte[] excelBytes = DocGen.GenerateExcelSpecification(matchedComponents, loadedConfig);

            // Вызываем CAD через стратегию шкафа!
            var dxfSheets = currentStrategy.GenerateCadDrawings(matchedComponents, input, CadGen);

            using var zipStream = new MemoryStream();
            using (var archive = new System.IO.Compression.ZipArchive(zipStream, System.IO.Compression.ZipArchiveMode.Create, true))
            {
                var excelEntry = archive.CreateEntry($"Спецификация_КП{loadedConfig.KpNumber}.xlsx");
                using (var entryStream = excelEntry.Open()) { entryStream.Write(excelBytes, 0, excelBytes.Length); }

                foreach (var sheet in dxfSheets)
                {
                    var dxfEntry = archive.CreateEntry(sheet.Key);
                    using (var entryStream = dxfEntry.Open()) { entryStream.Write(sheet.Value, 0, sheet.Value.Length); }
                }
            }
            zipStream.Position = 0;
            byte[] zipBytes = zipStream.ToArray();

            string archiveName = $"Проект_ШУВ_КП{loadedConfig.KpNumber}.zip";

            using var fileStream = new MemoryStream(zipBytes);
            using var streamRef = new DotNetStreamReference(stream: fileStream);
            await JS.InvokeVoidAsync("downloadFileFromStream", archiveName, streamRef);
        }
        catch (Exception ex)
        {
            System.Diagnostics.Debug.WriteLine($"Ошибка генерации архива: {ex.Message}");
        }
    }

    private async Task DownloadTkp()
    {
        if (loadedConfig == null || matchedComponents == null || !matchedComponents.Any()) return;

        try
        {
            var input = new UiConfigInput { EnclosureType = enclosureType, VoltageType = voltageType, TechnologyType = heaterType, ProjectNumber = loadedConfig.KpNumber };
            ICabinetStrategy currentStrategy = StrategyFactory.GetStrategy(selectedCabinetType);

            // Вызываем расчет ТКП через стратегию шкафа!
            CommercialProposal realProposal = currentStrategy.CalculateProposal(matchedComponents, input, marginMultiplier, PriceCalc);

            byte[] pdfBytes = DocGen.GenerateTkpPdf(realProposal);
            if (pdfBytes == null || pdfBytes.Length == 0) return;

            string fileName = $"ТКП_ШУВ_КП{loadedConfig.KpNumber}.pdf";

            using var fileStream = new MemoryStream(pdfBytes);
            using var streamRef = new DotNetStreamReference(stream: fileStream);
            await JS.InvokeVoidAsync("downloadFileFromStream", fileName, streamRef);
        }
        catch (Exception ex)
        {
            System.Diagnostics.Debug.WriteLine($"Ошибка генерации ТКП: {ex.Message}");
        }
    }
}
